function liral(elementId, lakk) { // 1. Xool ndax synthesis bi mu ngi dox ci phone bi if (!('speechSynthesis' in window)) { alert("Sa phone mënul jëfandikoo TTS bi. Jaaral ko ci Google Speech Services."); return; } // 2. Jël mbind mi let mbindMi = document.getElementById(elementId).innerText; // Nañu téye kàddu yiy wax bu jëkk ngir mu baña tàllal yaqule window.speechSynthesis.cancel(); let kaddoguTallah = new SpeechSynthesisUtterance(mbindMi); kaddoguTallah.lang = lakk; kaddoguTallah.rate = 0.75; // Yeexal ko ngir mu leer kaddoguTallah.pitch = 1.0; // 3. Wut kàddu gu dëppoo dëgg dëgg gu nekk ci biir phone bi let voices = window.speechSynthesis.getVoices(); let voiceFound = voices.find(v => v.lang.startsWith(lakk)); if (voiceFound) { kaddoguTallah.voice = voiceFound; } // 4. Lu am solo: xool bu amé erreur kaddoguTallah.onerror = function(event) { alert("Erreur TTS: " + event.error); }; // 5. Wax ko mu liral window.speechSynthesis.speak(kaddoguTallah); } // Loolu day fexé ba kàddu yi di charge ba fichiye bi di tàllalu if (typeof speechSynthesis !== 'undefined' && speechSynthesis.onvoiceschanged !== undefined) { speechSynthesis.onvoiceschanged = window.speechSynthesis.getVoices; }
